        // --- Main Animation Loop ---

        function gameLoop(timestamp) {

            const now = performance.now();

            if (lastTimestamp === 0) { lastTimestamp = now; }

            const deltaTime = Math.max(0, (now - lastTimestamp) / 1000);

            lastTimestamp = now;


            // Delayed Circle Creation

            if (now - lastCircleTime > circleInterval) {

                for (let i = 0; i < mouseHistory.length; i++) {

                    const entry = mouseHistory[i];

                    if (now - entry.time >= creationDelayMs) {

                        createCircle(entry.x, entry.y);

                        lastCircleTime = now;

                        mouseHistory.splice(0, i + 1);

                        break;

                    } else {

                        break;

                    }

                }

            }


            // History Cleanup

             if (mouseHistory.length > 0 && now - mouseHistory[0].time > creationDelayMs + 150) {

                mouseHistory = mouseHistory.filter(entry => now - entry.time < creationDelayMs + 150);

             }


            // Circle Movement & Collision

            if (boundaryRadiusPx > 0) {

                for (let [id, data] of circlesData) {

                    const circleElement = data.element;

                    if (!circleElement || !circleElement.parentNode) { circlesData.delete(id); continue; }

                    if (data.stopped) { continue; }


                    let directionX, directionY;

                    if (data.moveOutward) {

                        directionX = data.x - canvasCenterX;

                        directionY = data.y - canvasCenterY;

                    } else {

                        directionX = canvasCenterX - data.x;

                        directionY = canvasCenterY - data.y;

                    }


                    const distanceToCenter = Math.sqrt(directionX * directionX + directionY * directionY);

                    let normalizedVx = 0, normalizedVy = 0;


                    if (distanceToCenter > 1) {

                       normalizedVx = directionX / distanceToCenter;

                       normalizedVy = directionY / distanceToCenter;

                    } else if (!data.moveOutward) {

                        data.stopped = true; continue;

                    }


                    let nextX = data.x + normalizedVx * deltaTime * floatSpeedFactor;

                    let nextY = data.y + normalizedVy * deltaTime * floatSpeedFactor;

                    const nextDx = nextX - canvasCenterX, nextDy = nextY - canvasCenterY;

                    const nextDist = Math.sqrt(nextDx * nextDx + nextDy * nextDy);


                    if ((!data.moveOutward && nextDist <= boundaryRadiusPx) || (data.moveOutward && nextDist >= boundaryRadiusPx)) {

                        const angle = Math.atan2(nextDy, nextDx); // Angle to collision point

                        data.x = canvasCenterX + boundaryRadiusPx * Math.cos(angle); // Clamp to edge

                        data.y = canvasCenterY + boundaryRadiusPx * Math.sin(angle);

                        data.stopped = true;

                        if (circleElement) { circleElement.classList.add('stopped-scaled'); } // Trigger scale up


                        // *** Trigger Sound on Collision ***

                        if (audioStarted) {

                            // Map angle (-PI to PI) to frequency (minFrequency to maxFrequency)

                            // Normalize angle to 0-1 range first: (angle + Math.PI) / (2 * Math.PI)

                            const normalizedAngle = (angle + Math.PI) / (2 * Math.PI);

                            const frequency = minFrequency + normalizedAngle * frequencyRange;

                            try {

                                synth.triggerAttackRelease(frequency, "16n"); // Play a short note (16th note)

                            } catch (e) {

                                console.error("Tone.js error:", e);

                                // Potentially stop trying to play if context is lost

                            }

                        }


                        if (!data.particlesGenerated) {

                            let overlapCount = 0;

                            for (let [otherId, otherData] of circlesData) {

                                if (id !== otherId && otherData.stopped) {

                                    const dxOther = data.x - otherData.x;

                                    const dyOther = data.y - otherData.y;

                                    if (Math.sqrt(dxOther*dxOther + dyOther*dyOther) <= overlapThreshold) {

                                        overlapCount++;

                                    }

                                }

                            }

                            if (overlapCount >= 3) {

                                createOverlapParticles(data.x, data.y);

                                data.particlesGenerated = true;

                            }

                        }

                    } else {

                        data.x = nextX;

                        data.y = nextY;

                    }


                    const elementWidth = circleElement.offsetWidth;

                    const elementHeight = circleElement.offsetHeight;

                    if (elementWidth > 0 && elementHeight > 0) {

                       circleElement.style.left = `${data.x - elementWidth / 2}px`;

                       circleElement.style.top = `${data.y - elementHeight / 2}px`;

                    }

                }

            }

            requestAnimationFrame(gameLoop);

        }

            // *** Clamp recorded mouse coordinates to viewport ***

            mouseX = Math.max(0, Math.min(event.clientX, window.innerWidth));

            mouseY = Math.max(0, Math.min(event.clientY, window.innerHeight));
